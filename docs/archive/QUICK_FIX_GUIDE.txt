━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  STEP 1: Get Your Connection String
  ────────────────────────────────────

  Open SSMS → Right-click your database → Properties

  Copy these values:
    ✓ Server Name (e.g., monsterasp-srv.database.windows.net)
    ✓ Database Name (e.g., FitCore_DB)
    ✓ Username (your SQL login)
    ✓ Password (your SQL password)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  STEP 2: Update appsettings.json
  ─────────────────────────────────

  File: C:\Users\USER\source\repos\ERP_Project1\ERP_Project1\appsettings.json

  Replace ENTIRE file with:

  {
    "Logging": {
      "LogLevel": {
        "Default": "Information",
        "Microsoft.EntityFrameworkCore": "Warning"
      }
    },
    "ConnectionStrings": {
      "TenantErp": "Server=YOUR_SERVER;Initial Catalog=YOUR_DB;Persist Security Info=False;User ID=YOUR_USER;Password=YOUR_PASSWORD;MultipleActiveResultSets=True;Encrypt=True;TrustServerCertificate=True;Connection Timeout=30;"
    }
  }

  Replace:
    YOUR_SERVER    → monsterasp-srv.database.windows.net
    YOUR_DB        → Your database name
    YOUR_USER      → Your SQL username
    YOUR_PASSWORD  → Your SQL password

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  STEP 3: Run Application
  ───────────────────────

  Open PowerShell and run:

  cd C:\Users\USER\source\repos\ERP_Project1\ERP_Project1
  dotnet run

  ✅ If successful: Window opens, shows "Loaded X members"
  ❌ If error: Check connection string in Step 2

  ─────────────────

  Q: "Cannot connect to database" error
  A: Check connection string in appsettings.json
     Verify server name, database, username, password in SSMS first

  Q: "A second operation was started" error
  A: Already fixed! This error should NOT appear anymore

  Q: Application shows no members
  A: Database might be empty. Run migrations or add test data manually.

  Q: Where to find connection details?
  A: SSMS Object Explorer → Right-click database → Properties
